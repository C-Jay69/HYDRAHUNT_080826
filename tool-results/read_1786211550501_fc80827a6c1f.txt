    90→function formatDuration(seconds: number): string {
    91→  const m = String(Math.floor(seconds / 60)).padStart(2, '0')
    92→  const s = String(seconds % 60).padStart(2, '0')
    93→  return `${m}:${s}`
    94→}
    95→
    96→function formatDate(dateStr: string): string {
    97→  const d = new Date(dateStr)
    98→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
    99→}

   379→// ─── Active Session View ────────────────────────────────────────────────────

   381→function ActiveSession({ sessionId, onBack }: { sessionId: string; onBack: () => void }) {
   382→  const [session, setSession] = useState<InterviewSession | null>(null)
   383→  const [messages, setMessages] = useState<InterviewMessage[]>([])
   384→  const [scores, setScores] = useState<InterviewScore[]>([])
   385→  const [input, setInput] = useState('')
   386→  const [sending, setSending] = useState(false)
   387→  const [elapsed, setElapsed] = useState(0)
   388→  const [ended, setEnded] = useState(false)
   389→  const [liveScore, setLiveScore] = useState(0)
   390→  const [ending, setEnding] = useState(false)
   391→
   392→  const messagesEndRef = useRef<HTMLDivElement>(null)
   393→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   394→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   395→
   396→  // Fetch session data
   397→  useEffect(() => {
   398→    async function load() {
   399→      try {
   400→        const res = await fetch(`/api/interviews/${sessionId}`)
   401→        if (res.ok) {
   402→          const data = await res.json()
   403→          setSession(data)
   404→          setMessages(data.messages || [])
   405→          setScores(data.scores || [])
   406→          if (data.status === 'completed') {
   407→            setEnded(true)
   408→            if (data.scores?.length) {
   409→              const total = data.scores.reduce((sum: number, s: InterviewScore) => sum + s.score, 0)
   410→              setLiveScore(total)
   411→            }
   412→          }
   413→        }
   414→      } catch {
   415→        // handle error
   416→      }
   417→    }
   418→    load()
   419→  }, [sessionId])
   420→
   421→  // Timer
   422→  useEffect(() => {
   423→    if (ended) {
   424→      if (timerRef.current) clearInterval(timerRef.current)
   425→      return
   426→    }
   427→    timerRef.current = setInterval(() => {
   428→      setElapsed((prev) => prev + 1)
   429→    }, 1000)
   430→    return () => {
   431→      if (timerRef.current) clearInterval(timerRef.current)
   432→    }
   433→  }, [ended])
   434→
   435→  // Auto-scroll
   436→  useEffect(() => {
   437→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   438→  }, [messages])
   439→
   440→  const handleSend = async () => {
   441→    const trimmed = input.trim()
   442→    if (!trimmed || sending || ended) return
   443→
   444→    const userMsg: InterviewMessage = {
   445→      id: `local-${Date.now()}`,
   446→      role: 'user',
   447→      content: trimmed,
   448→      createdAt: new Date().toISOString(),
   449→    }
   450→    const updatedMessages = [...messages, userMsg]
   451→    setMessages(updatedMessages)
   452→    setInput('')
   453→    setSending(true)
   454→
   455→    // Reset textarea height
   456→    if (textareaRef.current) {
   457→      textareaRef.current.style.height = 'auto'
   458→    }
   459→
   460→    try {
   461→      const history = updatedMessages.map((m) => ({ role: m.role, content: m.content }))
   462→      const res = await fetch(`/api/interviews/${sessionId}/chat?XTransformPort=3000`, {
   463→        method: 'POST',
   464→        headers: { 'Content-Type': 'application/json' },
   465→        body: JSON.stringify({ message: trimmed, history }),
   466→      })
   467→
   468→      if (res.ok) {
   469→        const data = await res.json()
   470→        const aiMsg: InterviewMessage = {
   471→          id: data.id || `ai-${Date.now()}`,
   472→          role: 'ai',
   473→          content: data.content || data.message || '',
   474→          score: data.score,
   475→          createdAt: new Date().toISOString(),
   476→        }
   477→        setMessages((prev) => [...prev, aiMsg])
   478→
   479→        // Update live score from accumulated user message scores
   480→        if (data.score != null) {
   481→          setLiveScore((prev) => prev + data.score)
   482→        }
   483→        if (data.accumulatedScore != null) {
   484→          setLiveScore(data.accumulatedScore)
   485→        }
   486→      }
   487→    } catch {
   488→      // handle error silently
   489→    } finally {
   490→      setSending(false)
   491→    }
   492→  }
   493→
   494→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   495→    if (e.key === 'Enter' && !e.shiftKey) {
   496→      e.preventDefault()
   497→      handleSend()
   498→    }
   499→  }
   500→
   501→  const handleEndSession = async () => {
   502→    setEnding(true)
   503→    try {
   504→      const res = await fetch(`/api/interviews/${sessionId}?XTransformPort=3000`, {
   505→        method: 'PUT',
   506→        headers: { 'Content-Type': 'application/json' },
   507→        body: JSON.stringify({ status: 'completed' }),
   508→      })
   509→      if (res.ok) {
   510→        const data = await res.json()
   511→        setSession(data)
   512→        setScores(data.scores || [])
   513→        setEnded(true)
   514→        if (data.scores?.length) {
   515→          const total = data.scores.reduce((sum: number, s: InterviewScore) => sum + s.score, 0)
   516→          setLiveScore(total)
   517→        }
   518→      }
   519→    } catch {
   520→      // handle error
   521→    } finally {
   522→      setEnding(false)
   523→    }
   524→  }
   525→
   526→  const avgScore = ended && scores.length > 0
   527→    ? Math.round(scores.reduce((sum, s) => sum + s.score, 0) / scores.length * 10) / 10
   528→    : liveScore
   529→
   530→  return (
   531→    <div className="flex flex-col h-full gap-4">
   532→      {/* Top Bar */}
   533→      <div className="flex items-center justify-between gap-4">
   534→        <div className="flex items-center gap-3 min-w-0">
   535→          <Button
   536→            variant="ghost"
   537→            size="icon"
   538→            onClick={onBack}
   539→            className="shrink-0 text-hydra-muted hover:text-foreground"
   540→          >
   541→            <ArrowLeft className="h-4 w-4" />
   542→          </Button>
   543→          <div className="min-w-0">
   544→            <div className="flex items-center gap-2 flex-wrap">
   545→              <h2 className="text-lg font-semibold truncate">
   546→                {session?.role || 'Interview Session'}
   547→              </h2>
   548→              {session?.company && (
   549→                <span className="text-sm text-hydra-muted">@ {session.company}</span>
   550→              )}
   551→            </div>
   552→            <div className="flex items-center gap-2 mt-0.5">
   553→              <Badge
   554→                variant="outline"
   555→                className={`text-xs ${TYPE_COLORS[session?.type as SessionType] || ''}`}
   556→              >
   557→                {session?.type || 'behavioral'}
   558→              </Badge>
   559→              <span className="text-xs text-hydra-muted flex items-center gap-1">
   560→                <Clock className="h-3 w-3" />
   561→                {formatDuration(elapsed)}
   562→              </span>
   563→            </div>
   564→          </div>
   565→        </div>
   566→        {!ended && (
   567→          <Button
   568→            variant="outline"
   569→            onClick={handleEndSession}
   570→            disabled={ending || messages.length === 0}
   571→            className="border-hydra-red/30 text-hydra-red hover:bg-hydra-red/10 hover:text-hydra-red shrink-0 gap-2"
   572→          >
   573→            {ending && <Loader2 className="h-4 w-4 animate-spin" />}
   574→            End Session

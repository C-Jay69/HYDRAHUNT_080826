   408→/*  EDITOR VIEW                                                                */

   410→
   411→function EditorView({ resumeId }: { resumeId: string }) {
   412→  const { setSelectedResume } = useAppStore()
   413→  const queryClient = useQueryClient()
   414→
   415→  /* ---- fetch resume ---- */
   416→  const { data: resume, isLoading } = useQuery<Resume>({
   417→    queryKey: ['resume', resumeId],
   418→    queryFn: () => fetch(`/api/resumes/${resumeId}`).then((r) => r.json()),
   419→  })
   420→
   421→  /* ---- local editable state ---- */
   422→  const [title, setTitle] = useState('')
   423→  const [sections, setSections] = useState<ResumeSection[]>([])
   424→  const [saving, setSaving] = useState(false)
   425→  const [savedAt, setSavedAt] = useState<Date | null>(null)
   426→  const autosaveTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   427→  const firstLoad = useRef(true)
   428→
   429→  /* ---- populate local state from fetched data ---- */
   430→  useEffect(() => {
   431→    if (resume) {
   432→      setTitle(resume.title)
   433→      setSections(resume.sections)
   434→      firstLoad.current = false
   435→    }
   436→  }, [resume])
   437→
   438→  /* ---- autosave ---- */
   439→  const saveResume = useCallback(() => {
   440→    setSaving(true)
   441→    fetch(`/api/resumes/${resumeId}`, {
   442→      method: 'PUT',
   443→      headers: { 'Content-Type': 'application/json' },
   444→      body: JSON.stringify({ title, sections }),
   445→    })
   446→      .then(() => {
   447→        queryClient.invalidateQueries({ queryKey: ['resumes'] })
   448→        queryClient.invalidateQueries({ queryKey: ['resume', resumeId] })
   449→        setSavedAt(new Date())
   450→      })
   451→      .finally(() => setSaving(false))
   452→  }, [title, sections, resumeId, queryClient])
   453→
   454→  const scheduleSave = useCallback(() => {
   455→    if (firstLoad.current) return
   456→    if (autosaveTimer.current) clearTimeout(autosaveTimer.current)
   457→    autosaveTimer.current = setTimeout(() => {
   458→      saveResume()
   459→    }, 2000)
   460→  }, [saveResume])
   461→
   462→  useEffect(() => {
   463→    scheduleSave()
   464→    return () => {
   465→      if (autosaveTimer.current) clearTimeout(autosaveTimer.current)
   466→    }
   467→  }, [title, sections, scheduleSave])
   468→
   469→  /* ---- section helpers ---- */
   470→  const updateSectionContent = (idx: number, content: string) => {
   471→    setSections((prev) =>
   472→      prev.map((s, i) => (i === idx ? { ...s, content } : s)),
   473→    )
   474→  }
   475→
   476→  const sensors = useSensors(
   477→    useSensor(PointerSensor, { activationConstraint: { distance: 8 } }),
   478→  )
   479→
   480→  const handleDragEnd = (event: DragEndEvent) => {
   481→    const { active, over } = event
   482→    if (over && active.id !== over.id) {
   483→      setSections((prev) => {
   484→        const oldIdx = prev.findIndex((s) => s.id === active.id)
   485→        const newIdx = prev.findIndex((s) => s.id === over.id)
   486→        const reordered = arrayMove(prev, oldIdx, newIdx)
   487→        return reordered.map((s, i) => ({ ...s, sortOrder: i }))
   488→      })
   489→    }
   490→  }
   491→
   492→  /* ---- derived data for preview ---- */
   493→  const summaryText =
   494→    sections.find((s) => s.type === 'summary')?.content ?? ''
   495→  const experienceEntries: ExperienceEntry[] = JSON.parse(
   496→    sections.find((s) => s.type === 'experience')?.content ?? '[]',
   497→  )
   498→  const educationEntries: EducationEntry[] = JSON.parse(
   499→    sections.find((s) => s.type === 'education')?.content ?? '[]',
   500→  )
   501→  const skillsList: string[] = JSON.parse(
   502→    sections.find((s) => s.type === 'skills')?.content ?? '[]',
   503→  )
   504→  const projectEntries: ProjectEntry[] = JSON.parse(
   505→    sections.find((s) => s.type === 'projects')?.content ?? '[]',
   506→  )
   507→
   508→  /* ---- ATS score (simple heuristic for demo) ---- */
   509→  const computeAtsScore = () => {
   510→    let score = 0
   511→    if (summaryText.length > 50) score += 20
   512→    if (experienceEntries.length > 0) score += 25
   513→    if (educationEntries.length > 0) score += 15
   514→    if (skillsList.length >= 5) score += 20
   515→    if (projectEntries.length > 0) score += 10
   516→    const totalBullets = experienceEntries.reduce(
   517→      (a, e) => a + e.bullets.split('\n').filter(Boolean).length,
   518→      0,
   519→    )
   520→    if (totalBullets >= 5) score += 10
   521→    return Math.min(score, 100)
   522→  }
   523→
   524→  const atsScore = computeAtsScore()
   525→
   526→  if (isLoading) {
   527→    return (
   528→      <div className="flex items-center justify-center h-64">
   529→        <Loader2 className="size-8 text-hydra-purple animate-spin" />
   530→      </div>
   531→    )
   532→  }
   533→
   534→  return (
   535→    <motion.div
   536→      initial={{ opacity: 0 }}
   537→      animate={{ opacity: 1 }}
   538→      className="flex flex-col h-[calc(100vh-3.5rem)]"
   539→    >
   540→      {/* ---- Top Bar ---- */}
   541→      <div className="flex items-center gap-3 px-4 py-3 border-b border-hydra-border bg-hydra-surface/80 backdrop-blur-sm shrink-0">
   542→        <Button
   543→          variant="ghost"
   544→          size="sm"
   545→          onClick={() => setSelectedResume(null)}
   546→          className="text-hydra-muted hover:text-white shrink-0"
   547→        >
   548→          <ArrowLeft className="size-4 mr-1" />
   549→          Back
   550→        </Button>
   551→
   552→        <Input
   553→          value={title}
   554→          onChange={(e) => setTitle(e.target.value)}
   555→          className="bg-hydra-surface-2 border-hydra-border text-white font-semibold max-w-xs"
   556→        />
   557→
   558→        <div className="flex-1" />
   559→
   560→        {/* ATS Score Meter */}
   561→        <div className="hidden sm:flex items-center gap-2 min-w-[160px]">
   562→          <span className="text-xs text-hydra-muted">ATS</span>
   563→          <Progress
   564→            value={atsScore}
   565→            className="h-2 w-24"
   566→          />
   567→          <span
   568→            className={`text-xs font-bold min-w-[2rem] text-right ${
   569→              atsScore >= 70
   570→                ? 'text-emerald-400'
   571→                : atsScore >= 40
   572→                  ? 'text-hydra-purple'
   573→                  : 'text-red-400'
   574→            }`}
   575→          >
   576→            {atsScore}%
   577→          </span>
   578→        </div>
   579→
   580→        <Button
   581→          size="sm"
   582→          onClick={() => {
   583→            if (autosaveTimer.current) clearTimeout(autosaveTimer.current)
   584→            saveResume()
   585→          }}
   586→          disabled={saving}
   587→          className="bg-hydra-purple hover:bg-hydra-purple/80 text-white shrink-0"
   588→        >
   589→          {saving ? (
   590→            <Loader2 className="size-4 mr-1 animate-spin" />
   591→          ) : (
   592→            <Save className="size-4 mr-1" />
   593→          )}
   594→          Save
   595→        </Button>
   596→      </div>
   597→
   598→      {/* ---- Split Pane ---- */}
   599→      <ResizablePanelGroup direction="horizontal" className="flex-1">
   600→        {/* LEFT: Editor */}
   601→        <ResizablePanel defaultSize={55} minSize={35}>
   602→          <div className="h-full overflow-y-auto custom-scrollbar p-4 space-y-3">
   603→            <DndContext
   604→              sensors={sensors}
   605→              collisionDetection={closestCenter}
   606→              onDragEnd={handleDragEnd}
   607→            >
   608→              <SortableContext
   609→                items={sections.map((s) => s.id)}
   610→                strategy={verticalListSortingStrategy}
   611→              >
   612→                <AnimatePresence>
   613→                  {sections.map((section, idx) => (
   614→                    <motion.div
   615→                      key={section.id}
   616→                      layout
   617→                      initial={{ opacity: 0, y: 10 }}
   618→                      animate={{ opacity: 1, y: 0 }}
   619→                      exit={{ opacity: 0, y: -10 }}
   620→                    >
   621→                      <SortableSectionCard section={section}>
   622→                        <SectionEditor
   623→                          section={section}
   624→                          index={idx}
   625→                          onUpdateContent={(content) =>
   626→                            updateSectionContent(idx, content)
   627→                          }
   628→                        />
   629→                      </SortableSectionCard>
   630→                    </motion.div>
   631→                  ))}
   632→                </AnimatePresence>
   633→              </SortableContext>
   634→            </DndContext>
   635→          </div>
   636→        </ResizablePanel>
   637→
   638→        {/* Resize Handle */}
   639→        <ResizableHandle className="bg-hydra-border w-1 hover:bg-hydra-purple/40 transition-colors" />
   640→
   641→        {/* RIGHT: Live Preview */}
   642→        <ResizablePanel defaultSize={45} minSize={25}>
   643→          <div className="h-full overflow-y-auto custom-scrollbar p-4 bg-[#0a0612]">
   644→            <div className="max-w-[600px] mx-auto">
   645→              <ResumePreview
   646→                title={title}
   647→                summary={summaryText}

   649→                education={educationEntries}

   653→            </div>
